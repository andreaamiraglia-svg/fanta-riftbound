// Deck sharing format: versioned IDs preserve distinct cards with similar names.
function readDeckPayload(payload){
 if(!payload||typeof payload!=='object'||Array.isArray(payload)||payload.v!==1)throw new Error('Versione del codice mazzo non supportata.');
 for(const [kind,count] of Object.entries({champions:2,cards:18,monsters:12})){
  const ids=payload[kind];
  if(!Array.isArray(ids)||ids.length!==count||ids.some(id=>typeof id!=='string')||new Set(ids).size!==count)throw new Error('Servono 2 Campioni, 18 carte e 12 Mostri, tutti diversi.');
 }
 if(!validate(payload))throw new Error('Il mazzo contiene carte sconosciute o colori non consentiti dai Campioni.');
 return {name:typeof payload.name==='string'?payload.name.trim().slice(0,40)||'Mazzo importato':'Mazzo importato',...stripDeck(payload)};
}
function encodeDeck(d,name){
 const payload={v:1,name:String(name||deckName(d)||'Mazzo').trim().slice(0,40),...stripDeck(d)};
 readDeckPayload(payload);
 const bytes=new TextEncoder().encode(JSON.stringify(payload));
 return 'COTS1:'+btoa(Array.from(bytes,b=>String.fromCharCode(b)).join('')).replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');
}
function decodeDeck(code){
 const text=String(code||'').trim();
 if(!text||text.length>16000)throw new Error('Incolla un codice mazzo valido (massimo 16000 caratteri).');
 let payload;
 try{
  if(text.startsWith('{'))payload=JSON.parse(text);
  else{
   if(!/^COTS1:[A-Za-z0-9_-]+$/.test(text))throw new Error();
   let encoded=text.slice(6).replace(/-/g,'+').replace(/_/g,'/');
   encoded+='='.repeat((4-encoded.length%4)%4);
   const bytes=Uint8Array.from(atob(encoded),c=>c.charCodeAt(0));
   payload=JSON.parse(new TextDecoder('utf-8',{fatal:true}).decode(bytes));
  }
 }catch{throw new Error('Codice mazzo non valido o incompleto.');}
 return readDeckPayload(payload);
}
